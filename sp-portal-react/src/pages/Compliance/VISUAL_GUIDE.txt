└──────────────────────────────────────────────────────────────────────────┘

    useComplianceState() 
            │
            ├─► profiles: UserProfile[]
            │   ├─ id, name, email, phone
            │   ├─ vendor, role
            │   ├─ documents: ComplianceDocument[]
            │   ├─ trainings: Training[]
            │   ├─ vettingStatus: 'pending'|'in-progress'|'completed'|'rejected'
            │   └─ complianceScore: 0-100
            │
            ├─► vettings: VettingRecord[]
            │   ├─ id, profileId, status
            │   ├─ backgroundCheck: { criminalRecord, status, notes }
            │   ├─ checklist: ChecklistItem[]
            │   ├─ assignedOfficer, priority
            │   ├─ daysInStage, slaBreached
            │   └─ vendorAccess: 'full'|'restricted'|'none'
            │
            ├─► activeTab: 'profiles' | 'vetting'  ◄─── persisted to localStorage
            │
            ├─► selectedProfile: UserProfile | null
            ├─► selectedVetting: VettingRecord | null
            │
            ├─► modals: {
            │   ├─ profileDetail: boolean
            │   ├─ vettingDetail: boolean
            │   └─ documentUpload: boolean
            │   }
            │
            ├─► filters: {
            │   ├─ profiles: { status, vendor, role, searchQuery, sortBy, sortOrder }
            │   └─ vetting: { status, officer, priority, slaBreached, ... }
            │   }
            │
            ├─► loading: boolean
            └─► error: string | null

MAIN HOOK SIGNATURES:

  useComplianceState()
    → state: ComplianceState
    → selectProfile(profile): void
    → selectVetting(vetting): void
    → setActiveTab(tab): void
    → updateProfile(id, updates): void
    → updateVetting(id, updates): void
    → updateChecklistItem(vettingId, itemId, completed): void
    → [+ modal open/close functions]

  useConditionalRedirect({ vettings, onTabChange, ... })
    → handleOpenProfile(profile): void  ◄─ Main entry point
    → handleOpenVetting(vetting): void
    → isVettingPending(profileId): boolean
    → getVettingProgress(profileId): object

  useVettingStatus(vetting)
    → isComplete: boolean
    → isPending: boolean
    → progress: { completed, total, percentage }
    → pendingItems: ChecklistItem[]
    → nextStep: ChecklistItem
    → slaStatus: 'ok' | 'warning' | 'breached'


STATUS VALUES:

  ComplianceStatus (for profiles)
    'pending' | 'in-progress' | 'completed' | 'rejected'

  VettingStatus (for vettings)
    'pending' | 'in-progress' | 'completed' | 'rejected'

  DocumentStatus
    'pending' | 'approved' | 'rejected' | 'expired'

  TrainingStatus
    'not-started' | 'in-progress' | 'completed'


CUSTOM EVENTS:

  window.addEventListener('compliance-conditional-redirect', (e) => {
    e.detail: { profileId, vettingId, action, reason }
  })

  window.addEventListener('compliance-updated', (e) => {
    e.detail: { profileId, vettingId, status, vendorAccess, timestamp }
  })

  window.addEventListener('compliance-loaded', (e) => {
    e.detail: { totalProfiles, totalVettings, timestamp }
  })
